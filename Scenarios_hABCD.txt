scen=0

for n in "${n_values[@]}"; do
    ((scen++))
    filename="../../../Modularity_experiments/hABCD/scenA${scen}/same_size_${n}.txt"

    for i in $(seq 1 25); do \
	julia --project abcdh.jl -n "$n" -d 2.07,1,32 -c "$filename" \
		-x 0.37 -q 0.0,0.66,0.34 -w :strict -s "27$i" \
		-o "../../../Modularity_experiments/hABCD/scenA${scen}/rep${i}"
    done
done
